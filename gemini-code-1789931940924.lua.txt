	if on then
		TweenService:Create(Fill, TweenInfo.new(0.35), {Size = UDim2.new(1, 0, 1, 0)}):Play()
	else
		TweenService:Create(Fill, TweenInfo.new(0.25), {Size = UDim2.new(0, 0, 1, 0)}):Play()
	end
end

local dragStart, startPos
TopBar.InputBegan:Connect(function(input)
	if State.locked then return end
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		State.dragging = true
		dragStart = input.Position
		startPos = MainContainer.Position
	end
end)

UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		State.dragging = false
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if not State.dragging or State.locked then return end
	if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
		local delta = input.Position - dragStart
		MainContainer.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
	end
end)

-- ===================== AUTO STEAL =====================
local function trySteal()
	for _, name in ipairs({"Steal", "StealBrainrot", "Grab", "Take", "Collect", "StealItem", "StealObject"}) do
		local remote = ReplicatedStorage:FindFirstChild(name, true)
		if remote and remote:IsA("RemoteEvent") then
			pcall(function() remote:FireServer() end)
		end
	end
end

local lastSteal = 0
RunService.Heartbeat:Connect(function()
	if not State.enabled then return end

	local now = tick()
	local delay = 0.12

	if State.mode == "Semi" then
		delay = 0.32
	elseif State.mode == "Semi Normal" then
		if (now % 1) * 100 < State.semiPause then
			return
		end
		delay = 0.20
	end

	if now - lastSteal < delay then return end
	lastSteal = now

	trySteal()
end)

local frames, lastT = 0, tick()
RunService.RenderStepped:Connect(function()
	frames += 1
	if tick() - lastT >= 1 then
		TextLabel.Text = frames .. " FPS"
		frames = 0
		lastT = tick()
		local ping = math.floor(LocalPlayer:GetNetworkPing() * 1000)
		TextLabel2.Text = ping .. "ms"
	end
end)
